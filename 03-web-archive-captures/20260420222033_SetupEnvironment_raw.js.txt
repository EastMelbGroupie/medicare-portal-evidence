	function ShowDebug(spanobj,nme) {
		if (!document.getElementById(nme).checked)
			spanobj.innerHTML = '<input type="CheckBox" id="' + nme + '" name="' + nme + '"/>' + nme;
	}

	// Hide debug check box when mouse moves over span field
	function HideDebug(spanobj,nme) {
		if (!document.getElementById(nme).checked)
			spanobj.innerHTML = '<input type="Hidden" id="' + nme + '"name="' + nme + '"/>&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;';
	}

	// Toggle debug check box when mouse clicks on span field
	function ToggleDebug(nme) {
		if (document.getElementById(nme).checked == true)
			document.getElementById(nme).checked = false
		else
			document.getElementById(nme).checked = true;
	}
